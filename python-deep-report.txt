from fastapi import FastAPI, UploadFile, File, HTTPException
from minio import Minio
from dotenv import load_dotenv
from celery_app import process_uploaded_file, celery
import os
import uuid
import logging

load_dotenv("/opt/mxa/api/.env")

MINIO_ENDPOINT = os.getenv("MINIO_ENDPOINT")
MINIO_ACCESS_KEY = os.getenv("MINIO_ACCESS_KEY")
MINIO_SECRET_KEY = os.getenv("MINIO_SECRET_KEY")
MINIO_BUCKET = os.getenv("MINIO_BUCKET", "case-files")

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("mxa-api")

app = FastAPI(title="MxA Python Server", root_path="/api")

client = Minio(
    MINIO_ENDPOINT,
    access_key=MINIO_ACCESS_KEY,
    secret_key=MINIO_SECRET_KEY,
    secure=False,
)

@app.on_event("startup")
def startup():
    if not client.bucket_exists(MINIO_BUCKET):
        client.make_bucket(MINIO_BUCKET)

@app.get("/")
def root():
    return {"message": "MxA Python server is running"}

@app.get("/health")
def health():
    return {"status": "ok"}

@app.post("/upload")
async def upload(file: UploadFile = File(...)):
    try:
        object_name = f"{uuid.uuid4()}_{file.filename}"

        file.file.seek(0, 2)
        file_size = file.file.tell()
        file.file.seek(0)

        logger.info(f"Uploading file: {file.filename} | size={file_size} bytes")

        client.put_object(
            MINIO_BUCKET,
            object_name,
            data=file.file,
            length=file_size,
            content_type=file.content_type or "application/octet-stream",
            part_size=10 * 1024 * 1024,
        )

        task = process_uploaded_file.delay(object_name)

        return {
            "message": "File uploaded successfully",
            "object_name": object_name,
            "task_id": task.id,
            "size_bytes": file_size,
            "bucket": MINIO_BUCKET,
        }

    except Exception as e:
        logger.exception("Upload failed")
        raise HTTPException(status_code=500, detail=f"Upload failed: {str(e)}")

@app.get("/task/{task_id}")
def get_task_status(task_id: str):
    try:
        task_result = celery.AsyncResult(task_id)
        return {
            "task_id": task_id,
            "status": task_result.status,
            "result": task_result.result if task_result.ready() else None,
        }
    except Exception as e:
        logger.exception("Task lookup failed")
        raise HTTPException(status_code=500, detail=f"Task lookup failed: {str(e)}")
